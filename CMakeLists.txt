cmake_minimum_required(VERSION 3.20)
project(CloudBlockLocal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cloudblock src/main.cpp)
if(MSVC)
  target_compile_options(cloudblock PRIVATE /W4 /EHsc)
else()
  target_compile_options(cloudblock PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME cloudblock_local
           COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_local.py" "$<TARGET_FILE:cloudblock>" )
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(cloudblock-nbd src/nbd_server.cpp)
  target_compile_options(cloudblock-nbd PRIVATE -Wall -Wextra -Wpedantic)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cloudblock_nbd_protocol
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_nbd.py" "$<TARGET_FILE:cloudblock-nbd>")
  endif()
endif()

# Native Windows block-device adapter is optional. The core CLI builds without a driver.
option(CLOUDBLOCK_BUILD_WNBD "Build direct user-mode WNBD adapter" OFF)
if(WIN32 AND CLOUDBLOCK_BUILD_WNBD)
  if(NOT WNBD_INCLUDE_DIR OR NOT EXISTS "${WNBD_INCLUDE_DIR}/wnbd.h")
    message(FATAL_ERROR "Provide -DWNBD_INCLUDE_DIR=<path to the WNBD SDK include directory>")
  endif()
  add_executable(cloudblock-windows src/windows_wnbd.cpp)
  target_include_directories(cloudblock-windows PRIVATE "${WNBD_INCLUDE_DIR}")
  target_compile_definitions(cloudblock-windows PRIVATE UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(cloudblock-windows PRIVATE /W4 /EHsc /utf-8)
  else()
    target_compile_options(cloudblock-windows PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
